/*NB the extract contains all ACP episodes - we want to restrict to those that started with an unplanned/emergency admission
It is possible to be transferred from a planned admission in to the ED for resus for example - but these cases do not represent unplanned care thus we will use admimeth to restrict the sample to those with an emergency admission - see https://archive.datadictionary.nhs.uk/DD%20Release%20May%202024/data_elements/admission_method_code__hospital_provider_spell_.html.
We cannot restrict the ae data in the same way so, it may contain some cases where referral to ED came from within a planned admission in the hospital - important to note the differences in the methods - our ae data is all ae activity (some of which may have come from planned admissions ~5%s) versus our acp data is all unplanned*/

/*look for duplicates at episode level - none found*/
duplicates tag epikey, gen(tag1)
drop tag1


/*collapse to spell level - the data is currently at episode level AND final disdate has already been imputed so collapse to spell level using patid and admidate*/
/*first check that all vars i want to keep are consistent at spell level*/
browse person_id_deid cips_admidate cips_disdate ADMIMETH date_of_death age_at_arrival sex_code imd_quintile ethnicity_18_group ICB22CD_short
collapse (firstnm) cips_disdate ADMIMETH date_of_death age_at_arrival sex_code imd_quintile ethnicity_18_group ICB22CD_short, by (person_id_deid cips_admidate)


drop if cips_admidate==.

/*some spells have no start and end date*/
browse if cips_disdate==.
replace cips_disdate=date_of_death if cips_disdate==.
gen flag_disdate=1 if cips_disdate==.
tab flag_disdate, mi


drop if cips_disdate==. 
drop flag_disdate


keep if cips_disdate>=td(01jan2013)
keep if cips_admidate<=td(31dec2023)

/*flag if disdate before admidate*/
gen flag1=1 if cips_disdate<cips_admidate
tab flag1, mi
drop if flag1==1 
drop flag1

/*flag cases where arrival date comes after the date of death*/


gen flag=1 if cips_disdate > date_of_death & date_of_death!=.
tab flag, mi
gen days_flag= cips_disdate-date_of_death
sum days_flag if flag==1, detail
replace cips_disdate=date_of_death if flag==1 & (days_flag>=1 & days_flag<=7) 
drop flag
drop days_flag
/*same for admidate*/
gen flag=1 if cips_admidate > date_of_death & date_of_death!=.
tab flag, mi
drop if flag==1

/*destring vars to make file more managable*/
destring age_at_arrival, replace
encode ethnicity_18_group, gen(ethnicity18)
drop ethnicity_18_group

/*gen died within 1 year of admission date - (could have flagged people who died within a year of disdate but have decided it makes most sense to talk about people who died within a year of the admission date for admissions and for bed days)*/
gen days_between=date_of_death-cips_admidate
gen lyol_flag=0
replace lyol_flag=1 if days_between<=365
/*gen quart var*/
gen quarter = quarter(cips_admidate)
gen total=1
/*gen year*/
gen year = yofd(cips_admidate)

/*for the admissions summary need to refine the time period so admidate >=01/01/2013 - because atm we have people admitted before study start period so we can keep those day for the unplanned days count*/
gen study_period=1 if cips_admidate>=td(01jan2013)

/*collapse for ED visit numbers*/
preserve
keep if study_period==1
collapse (sum) total lyol_flag, by(year quarter)
/*round to nearest 5 for output*/
gen lyol_round = round(lyol_flag, 5)
gen total_round = round(total, 5)
drop lyol_flag total
/*save*/
export excel using "$output\ACP_2013_2023.xls", firstrow(varlabel) sheet(unplanned_acp) sheetreplace
restore

/*interaction models*/
/*imd unadjusted*/
poisson lyol_flag i.imd_quintile##i.time_period, vce(robust)
margins imd_quintile#time_period
marginsplot

/*imd - with margins at median age*/
poisson lyol_flag age_at_arrival i.sex_code i.imd_quintile##i.time_period, vce(robust)
estimates store imd_model
etable, estimates(imd_model) cstat(_r_b) cstat(_r_ci) cstat(_r_p) export("$output\imd_interaction.xls", replace)
margins imd_quintile#time_period, at(age=65)  
etable, margins cstat(_r_b) cstat(_r_ci) export("$output\imd_marginal_v2.xls", replace)
marginsplot, title("") xtitle("deprivation quintiles (1 is most deprived)") ytitle("predicted proportion")
graph export "$output\imd_marginal_v2.png", replace

/*imd - with margins at age 80*/
